    22→type PayWindow = Window & {
    23→  PaystackPop?: {
    24→    setup: (o: {
    25→      key: string;
    26→      email: string;
    27→      amount: number;
    28→      currency: string;
    29→      ref: string;
    30→      callback: (r: { reference: string }) => void;
    31→      onClose: () => void;
    32→    }) => { openIframe: () => void };
    33→  };
    34→  FlutterwaveCheckout?: (o: Record<string, unknown>) => void;
    35→  paypal?: {
    36→    Buttons: (o: Record<string, unknown>) => { render: (sel: string | HTMLElement) => Promise<void> };

    39→
    40→function loadScript(src: string, id: string): Promise<void> {
    41→  return new Promise((resolve, reject) => {
    42→    const existing = document.getElementById(id);
    43→    if (existing) return resolve();
    44→    const s = document.createElement("script");
    45→    s.src = src;
    46→    s.id = id;
    47→    s.async = true;
    48→    s.onload = () => resolve();
    49→    s.onerror = () => reject(new Error("Failed to load payment script"));
    50→    document.head.appendChild(s);
    51→  });
    52→}

   424→
   425→function AccountStep({ onAuthed }: { onAuthed: (u: StudioUser) => void }) {
   426→  const [mode, setMode] = useState<"signup" | "signin">("signup");
   427→  const [form, setForm] = useState({ name: "", email: "", password: "" });
   428→  const [busy, setBusy] = useState(false);
   429→  const [error, setError] = useState("");
   430→  const [googleReady, setGoogleReady] = useState(false);
   431→
   432→  useEffect(() => {
   433→    api<{ configured: boolean }>("/api/auth/google/status")
   434→      .then((r) => setGoogleReady(r.configured === true))
   435→      .catch(() => setGoogleReady(false));
   436→  }, []);
   437→
   438→  async function submit(e: React.FormEvent) {
   439→    e.preventDefault();
   440→    setBusy(true);
   441→    setError("");
   442→    try {
   443→      const url = mode === "signin" ? "/api/user/login" : "/api/user/register";
   444→      const body =
   445→        mode === "signin"
   446→          ? { email: form.email, password: form.password }
   447→          : { name: form.name, email: form.email, password: form.password };
   448→      const res = await api<{ user: StudioUser }>(url, { method: "POST", body: JSON.stringify(body) });
   449→      onAuthed(res.user);
   450→    } catch (err) {
   451→      setError(err instanceof Error ? err.message : "Something went wrong");
   452→    } finally {
   453→      setBusy(false);
   454→    }
   455→  }
   456→
   457→  const nextParam = encodeURIComponent("/#subscribe");
   458→
   459→  return (
   460→    <div className="border border-white/15 bg-white/5 p-6">
   461→      <h2 className="font-black uppercase tracking-tight text-lg">1 — Your account</h2>
   462→      <p className="mt-1 text-xs text-white/50">
   463→        {mode === "signup"
   464→          ? "Create it now — your plan attaches straight to it."
   465→          : "Sign in and we'll attach the plan to your account."}
   466→      </p>
   467→
   468→      <form onSubmit={submit} className="mt-5 space-y-4">
   469→        {mode === "signup" && (
